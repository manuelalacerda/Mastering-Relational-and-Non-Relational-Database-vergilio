CREATE OR REPLACE procedure prd_insert_alunos (p_id number, p_nome varchar2,p_turma varchar2,p_nota NUMBER)AS
BEGIN

    id,

    turma,
    nota
) VALUES ( p_id,
           p_nome,
           p_turma,
           p_nota );

end;

//

create function media_alunos(
nome VARCHAR
)return number is
    v_media NUMBER;
BEGIN
SELECT avg(nota)INTO v_media FROM alunos10 GROUP BY nome; 
return v_media;
end;
